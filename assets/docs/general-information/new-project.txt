### vite.config.js
import { defineConfig } from "vite";

export default defineConfig({
  server: {
    port: 3000,
    watch: {
      usePolling: true,
    },
  },
  plugins: [
    {
      name: "html-force-reload",
      handleHotUpdate({ file, server }) {
        if (file.endsWith(".html")) {
          server.ws.send({ type: "full-reload", path: "*" });
        }
      },
    },
  ],
});
